cmake_minimum_required(VERSION 3.20)
project(Samera VERSION 1.0.0 LANGUAGES CXX)
include(GNUInstallDirs)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets Multimedia)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GST REQUIRED IMPORTED_TARGET gstreamer-app-1.0>=1.20 gstreamer-video-1.0>=1.20)
add_executable(samera main.cpp)
target_link_libraries(samera PRIVATE Qt6::Widgets Qt6::Multimedia PkgConfig::GST)
target_compile_options(samera PRIVATE -Wall -Wextra)
install(TARGETS samera RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES data/samera.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
install(FILES data/samera.svg DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps)
install(FILES README.md LICENSE docs/USAGE.md docs/BUILD.md DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/samera)
install(PROGRAMS scripts/install-user.sh DESTINATION .)

set(CPACK_GENERATOR TGZ)
set(CPACK_PACKAGE_NAME samera)
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
set(CPACK_PACKAGE_FILE_NAME "samera-${PROJECT_VERSION}-linux-${CMAKE_SYSTEM_PROCESSOR}")
set(CPACK_PACKAGE_VENDOR xSayer)
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "A lightweight native camera app for Linux")
set(CPACK_INCLUDE_TOPLEVEL_DIRECTORY ON)
set(CPACK_STRIP_FILES ON)
set(CPACK_PACKAGING_INSTALL_PREFIX "/")
include(CPack)
